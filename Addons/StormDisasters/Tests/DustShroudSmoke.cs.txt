// Check the live dust wrap without touching Sandbox's spawn or undo stack.
var scene = Game.ActiveScene;
var p = new Vector3(5200f, 4300f, 16f);
if (scene.GetAllComponents<Prop>().Any(x => (x.WorldPosition - p).WithZ(0f).Length < 3500f))
    throw new Exception("Dust probe area is occupied by existing gameplay props.");
var results = new System.Collections.Generic.List<object>();
var failures = new System.Collections.Generic.List<string>();
foreach (var name in new[] { "rope_tornado", "funnel_tornado", "wedge_tornado" })
{
    var tornado = GameObject.GetPrefab($"storm_disasters/{name}.prefab").Clone(p);
    tornado.NetworkMode = NetworkMode.Never;
    try
    {
        ModelRenderer volume = null;
        for (var i = 0; i < 60 && (!volume.IsValid() || volume.Attributes.GetVector4("VortexPhase", default).w < 1.35f); i++)
        {
            await Task.Delay(50, cancellationToken);
            volume = tornado.Children.FirstOrDefault(o => o.Name == "Condensation Volume")?.GetComponent<ModelRenderer>();
        }
        if (!volume.IsValid()) throw new Exception($"{name}: missing the condensation volume.");
        var dust = volume.Attributes.GetVector4("VortexDustProfile", default);
        if (dust.x < 275f || dust.z < 2.55f)
            failures.Add($"{name}: missing the volume's original sandy dust blend.");
        var sprites = tornado.Children.Where(o => o.Name == "Tornado smoke").SelectMany(o => o.Children)
            .Count(o => o.GetComponent<SpriteRenderer>().IsValid());
        if (sprites != 0)
            failures.Add($"{name}: expected no separate dust overlay on a shader-backed funnel, got {sprites}.");
        var firstPhase = volume.Attributes.GetVector4("VortexPhase", default);
        await Task.Delay(200, cancellationToken);
        if (volume.Attributes.GetVector4("VortexPhase", default).w <= firstPhase.w)
            failures.Add($"{name}: the volume no longer animates.");
        results.Add(new { name, Sprites = sprites, DustHeight = dust.x, DustDensity = dust.z });
    }
    finally { if (tornado.IsValid()) tornado.Destroy(); }
}
if (failures.Count > 0) throw new Exception(string.Join(" ", failures));
return results;
