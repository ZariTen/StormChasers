// Run on Flatgrass with no other tornadoes. Probe near-ground rendering and local F5 rain.
var scene = Game.ActiveScene;
if (!scene.GetAllComponents<MapInstance>().Any(m => m.MapName == "facepunch.flatgrass")) throw new Exception("Expected Flatgrass.");
if (scene.GetAllObjects(true).Any(o => o.Tags.Has("storm_disaster_tornado"))) throw new Exception("Run with no other tornadoes.");
var view = scene.Camera;
var start = view.WorldPosition + view.WorldRotation.Forward.WithZ(0).Normal * 3300f;
if (scene.GetAllComponents<Prop>().Any(p => (p.WorldPosition - start).WithZ(0).Length < 5200f)) throw new Exception("Probe area occupied by props.");
var ground = scene.Trace.Ray(start + Vector3.Up * 1200f, start - Vector3.Up * 2000f).WithoutTags("player", "trigger").IgnoreDynamic().IgnoreKeyframed().Run();
if (!ground.Hit || ground.Normal.z < .95f) throw new Exception("No probe ground.");
var tornado = GameObject.GetPrefab("storm_disasters/wedge_tornado.prefab").Clone(ground.HitPosition);
tornado.NetworkMode = NetworkMode.Never;
var cameraObject = new GameObject(true, "Wedge surface probe camera");
cameraObject.NetworkMode = NetworkMode.Never;
try
{
    var camera = cameraObject.AddComponent<CameraComponent>();
    camera.IsMainCamera = false;
    foreach (var tag in new[] { "player", "viewmodel", "storm_disasters_weather", "wedge_probe_debris" }) camera.RenderExcludeTags.Add(tag);
    camera.FieldOfView = 44f;
    camera.ZFar = 16000f;
    ModelRenderer volume = null;
    for (int i = 0; i < 450; i++)
    {
        await Task.Delay(36, cancellationToken);
        volume = tornado.Children.FirstOrDefault(o => o.Name == "Condensation Volume")?.GetComponent<ModelRenderer>();
        if (volume.IsValid() && volume.Attributes.GetVector4("VortexPhase", default).w >= 12.25f) break;
    }
    if (!volume.IsValid() || volume.Attributes.GetVector4("VortexPhase", default).y > -.115f) throw new Exception("Expected a mature wedge.");
    var origin = tornado.WorldPosition;
    var support = scene.Trace.Ray(origin + Vector3.Up * 96f, origin - Vector3.Up * 96f).WithoutTags("player", "trigger").Run();
    if (!support.Hit || MathF.Abs(origin.z - support.HitPosition.z) > 2f) throw new Exception("Wedge root is not grounded.");
    tornado.Children.First(o => o.Name == "Airborne Debris").Tags.Add("wedge_probe_debris");
    cameraObject.WorldPosition = origin + new Vector3(-3100f, 0, 65f);
    cameraObject.WorldRotation = Rotation.Identity;
    async Task<Color[]> Capture(bool visible)
    {
        volume.Enabled = visible;
        await Task.Delay(70, cancellationToken);
        using var bitmap = new Bitmap(768, 420);
        camera.RenderToBitmap(bitmap, false);
        return new[] { 195, 207, 212, 217, 221, 224 }.SelectMany(y => Enumerable.Range(362, 45).Select(x => bitmap.GetPixel(x, y))).ToArray();
    }
    var on = await Capture(true);
    var off = await Capture(false);
    volume.Enabled = true;
    var bands = on.Zip(off, (a,b) => (MathF.Abs(a.r-b.r)+MathF.Abs(a.g-b.g)+MathF.Abs(a.b-b.b))/3f).Chunk(45).Select(a => a.Average()).ToArray();
    // Read visible streak length from the same prefab's actual local weather at the wall and far away.
    async Task<float> RainAt(float distance)
    {
        var target = view.WorldPosition + view.WorldRotation.Forward.WithZ(0).Normal * distance;
        var floor = scene.Trace.Ray(target + Vector3.Up * 1000f, target - Vector3.Up * 2000f).WithoutTags("player", "trigger").Run();
        if (!floor.Hit) throw new Exception("Could not position weather probe.");
        tornado.WorldPosition = floor.HitPosition;
        await Task.Delay(380, cancellationToken);
        var drops = tornado.Children.Single(o => o.Name == "Storm Disasters Weather").Children.Single(o => o.Name == "Nearby Rain").Children
            .Select(o => o.GetComponent<SpriteRenderer>()).Where(r => r.IsValid() && r.Enabled && r.Color.a > .08f).Select(r => r.Size.y).ToArray();
        if (drops.Length == 0) throw new Exception("No outdoor rain sampled.");
        return drops.Average();
    }
    var wallRain = await RainAt(2500f);
    var farRain = await RainAt(7200f);
    var evidence = $"Ground contact bands [{string.Join(", ", bands.Select(x => x.ToString("F3")))}]; average rain streak {wallRain:F1} near wall, {farRain:F1} far away.";
    // Ground-adjacent rows must show the skirt; foreground grass may remain clear.
    if (bands[3] < .09f || bands[4] < .04f || wallRain < farRain * 1.65f) throw new Exception(evidence);
    return evidence;
}
finally { if (tornado.IsValid()) tornado.Destroy(); if (cameraObject.IsValid()) cameraObject.Destroy(); }
