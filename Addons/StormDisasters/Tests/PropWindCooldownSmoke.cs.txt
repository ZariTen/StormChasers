// Run on a private local Flatgrass host. Allow 60s: the ratings run one at a time in the same clear area.
var scene = Game.ActiveScene;
if (!Networking.IsHost || !scene.GetAllComponents<MapInstance>().Any(x => x.MapName == "facepunch.flatgrass"))
    throw new Exception("Expected local host on stock Flatgrass.");
var probes = new[] {
    (Rating: 3, Seconds: 20f, Prefab: "cylinder_tornado_f3", R: 295f),
    (Rating: 4, Seconds: 15f, Prefab: "wedge_tornado_f4", R: 625f),
    (Rating: 5, Seconds: 10f, Prefab: "wedge_tornado", R: 850f),
};
// The F5 wedge's broad-phase prop scan is 7,922 units. Allow for small travel between resets.
const float clearance = 8200f;
bool AreaClear(Vector3 point) =>
    scene.GetAllComponents<Prop>().All(p => (p.WorldPosition - point).WithZ(0f).Length > clearance) &&
    scene.GetAllComponents<PlayerController>().All(p => (p.WorldPosition - point).WithZ(0f).Length > clearance);
void RequireNoTornadoes()
{
    if (scene.GetAllObjects(true).Any(o => o.Tags.Has("storm_disaster_tornado")))
        throw new Exception("Remove existing tornadoes before running the cooldown probe.");
}
RequireNoTornadoes();
var point = new[] { new Vector3(6700,6200,0), new Vector3(-6700,-6200,0), new Vector3(6700,-6200,0), new Vector3(-6700,6200,0) }
    .FirstOrDefault(AreaClear);
if (point == Vector3.Zero) throw new Exception("No probe area clear of pre-existing props and players.");
var floor = scene.Trace.Ray(point + Vector3.Up * 100f, point - Vector3.Up * 100f).IgnoreDynamic().IgnoreKeyframed().Run();
if (!floor.Hit || floor.Normal.z < 0.65f) throw new Exception("No Flatgrass floor for cooldown probe.");
var origin = floor.HitPosition;
const int samplesPerRing = 24;
const int sample = 2 * samplesPerRing;
var results = new System.Collections.Generic.List<object>();
var model = Model.Load("models/citizen_props/crate01.vmdl");
if (model.IsError || model.Physics == null) throw new Exception("No stock crate physics.");
foreach (var probe in probes)
{
    RequireNoTornadoes();
    if (!AreaClear(origin)) throw new Exception($"F{probe.Rating} probe area is no longer clear.");
    var crates = new GameObject[sample];
    GameObject root = null;
    try
    {
        for (int j = 0; j < sample; j++)
        {
            var crate = scene.CreateObject(false);
            crates[j] = crate;
            crate.Name = "Wind: cooldown test crate";
            crate.NetworkMode = NetworkMode.Never;
            float angle = j * MathF.Tau / samplesPerRing;
            crate.WorldPosition = origin + new Vector3(MathF.Cos(angle) * probe.R, MathF.Sin(angle) * probe.R, 80f + 100f * (j / samplesPerRing));
            crate.LocalScale = Vector3.One * 0.2f;
            crate.Components.Create<Prop>().Model = model;
            crate.Enabled = true;
        }
        await Task.Delay(180, cancellationToken);
        var bodies = crates.Select(o => o.GetComponent<Rigidbody>()).ToArray();
        if (bodies.Any(b => !b.IsValid())) throw new Exception("Probe crates did not create rigidbodies.");
        foreach (var body in bodies) { body.Gravity = false; body.Velocity = Vector3.Zero; body.MotionEnabled = false; }
        await Task.Delay(180, cancellationToken);
        if (bodies.Any(body => body.MotionEnabled)) throw new Exception("Probe crates failed to freeze.");

        var started = Time.Now;
        root = GameObject.GetPrefab($"storm_disasters/{probe.Prefab}.prefab").Clone(origin);
        root.NetworkMode = NetworkMode.Never;
        int released = 0;
        int initial = -1;
        while (true)
        {
            await Task.Delay(40, cancellationToken);
            var elapsed = Time.Now - started;
            if (!root.IsValid()) throw new Exception($"F{probe.Rating} disappeared.");
            root.WorldPosition = origin;
            // Don't let freed test props fly across the live playtest.
            for (int j = 0; j < sample; j++)
            {
                var body = bodies[j];
                if (!body.IsValid() || !body.MotionEnabled) continue;
                released++;
                crates[j].Destroy();
                bodies[j] = null;
            }
            if (elapsed < 0.95f) continue;
            if (initial < 0)
            {
                if (released == sample) throw new Exception($"F{probe.Rating} left no frozen crates to observe.");
                initial = released;
            }
            else if (released > initial)
            {
                if (elapsed < probe.Seconds)
                    throw new Exception($"F{probe.Rating} retried too early: {elapsed:F2}s.");
                results.Add(new { probe.Rating, probe.Seconds, FirstObservedRetry = elapsed, InitialUnfrozen = initial, AfterRetry = released, Sample = sample });
                break;
            }
            else if (elapsed > probe.Seconds + 1.5f)
                throw new Exception($"F{probe.Rating} did not unfreeze more crates after the {probe.Seconds}s cooldown. Initial={initial}.");
        }
    }
    finally
    {
        if (root.IsValid()) root.Destroy();
        foreach (var crate in crates) if (crate.IsValid()) crate.Destroy();
    }
    // Let destroyed objects leave the scene before checking the next trial's clearance.
    await Task.Delay(40, cancellationToken);
}
return results;
