var scene=Game.ActiveScene;
var test=scene.CreateObject(false);
const string tag="wedge_smoke_probe";
test.Name="F5 geometry verification";
test.NetworkMode=NetworkMode.Never;
test.Tags.Add(tag);
test.WorldPosition=new Vector3(-5000,4900,0);
var camera=scene.Camera;
if(camera.IsValid()) camera.RenderExcludeTags.Add(tag);
try
{
    var funnel=test.Components.Create<StormTornadoFunnel>();
    funnel.Rating=5;
    test.Enabled=true;
    await Task.Delay(500);
    var body=test.Children.Single(o=>o.Name=="Condensation Volume").GetComponent<ModelRenderer>();
    var bound=body.Attributes.GetVector4("VortexBounds",default);
    var radii=body.Attributes.GetVector4("VortexRadii",default);
    var clearance=bound.w;
    if(clearance<430 || radii.x<1400 || radii.x-clearance<850 || radii.y<=radii.x || body.Bounds.Maxs.z<5000)
        throw new Exception($"F5 needs a walk-in hollow core and a much broader wall, got {clearance:F0} units clear and a {radii.x:F0} ground radius.");
    if(test.Components.GetAll<Collider>(FindMode.EverythingInSelfAndDescendants).Any())
        throw new Exception("Hollow eye tested with unexpected blocking collision.");
    var crown=body.Attributes.GetVector4("StormCloudBounds",default);
    if (crown.x<3100 || crown.z<740 || test.Children.Any(o=>o.Name=="Eye Crown"))
        throw new Exception("The F5 crown should be integrated into the same volume, without an overhead card.");
    var weather=scene.GetAllComponents<StormWeather>().Single();
    var deck=weather.GameObject.Children.Single(o=>o.Name=="Overcast Deck").Children.Select(o=>o.GetComponent<SpriteRenderer>()).ToArray();
    if(deck.Count(r=>r.Size.x>=40000 && r.Size.y>=30000)<2 || deck.Max(r=>r.WorldPosition.z)-deck.Min(r=>r.WorldPosition.z)<1200)
        throw new Exception("Persistent cloud strata should extend far beyond the tornado and have distinct heights.");
    return new{Passed=true,Clearance=clearance,GroundRadius=radii.x,DeckSpan=deck.Max(r=>r.Size.x)};
}
finally
{
    test.Destroy();
    if(camera.IsValid()) camera.RenderExcludeTags.Remove(tag);
}
